; Examples for Tiled maps and sprite sheets. Format: see tools/check_doc_examples.py.

=== LoadTiledMap
; a level drawn in the Tiled editor
Graphics 800, 480, 0, 2
map = LoadTiledMap("examples/media/tiled/meadow.tmx")
If map = 0 Then RuntimeError "meadow.tmx not found"

=== FreeTiledMap
; a game with two levels: free each map before loading the next
For level = 1 To 2
    map = LoadTiledMap("examples/media/tiled/meadow.tmx")
    Print "Level " + level + " has " + CountTiledObjects(map, "coin") + " coins"   ; 8 coins
    FreeTiledMap map                  ; the map, its layers, objects and picture are freed
Next

=== DrawTiledMap
; every visible layer, scrolled by the 2D camera
map = LoadTiledMap("examples/media/tiled/meadow.tmx")
Camera2D 480, 240
DrawTiledMap map

=== TiledMapImage
; the picture holding every tile, for drawing tile objects
map = LoadTiledMap("examples/media/tiled/meadow.tmx")
coin = TiledObject(map, 1, "coin")
DrawImage TiledMapImage(map), TiledObjectX(coin), TiledObjectY(coin), TiledObjectTile(coin)

=== TiledMapProperty
; read the map's own properties, with a fallback for ones the map does not set
map = LoadTiledMap("examples/media/tiled/meadow.tmx")
title$ = TiledMapProperty(map, "title", "Untitled")
music$ = TiledMapProperty(map, "music", "none")
Print title$                          ; Meadow Run
Print music$                          ; none

=== CountTiledLayers

=== TiledLayerName
; list the layers
map = LoadTiledMap("examples/media/tiled/meadow.tmx")

    Print TiledLayerName(map, i)

=== TiledLayer
; a layer is a tile map: make the ground solid
map = LoadTiledMap("examples/media/tiled/meadow.tmx")
ground = TiledLayer(map, "Ground")
TileMapBody2D ground

=== ShowTiledLayer
; draw a level, and let the player switch the Decor layer off and on with D
Graphics 960, 480
map = LoadTiledMap("examples/media/tiled/meadow.tmx")
show = True
While Not KeyHit(KEY_ESCAPE)
    If KeyHit(KEY_D) Then show = Not show
    ShowTiledLayer map, "Decor", show
    Cls
    DrawTiledMap map
    Flip
Wend

=== CountTiledObjects
; how many coins are in the level
map = LoadTiledMap("examples/media/tiled/meadow.tmx")
Print CountTiledObjects(map, "coin")

=== TiledObject
; walk through all the coins of a map and read the value typed on each in Tiled
map = LoadTiledMap("examples/media/tiled/meadow.tmx")
For i = 1 To CountTiledObjects(map, "coin")
    coin = TiledObject(map, i, "coin")
    Print "coin " + i + " is worth " + TiledObjectProperty(coin, "value")
Next                                  ; the first line is: coin 1 is worth 10

=== FindTiledObject
; find the level's goal by the name given to it in Tiled
map = LoadTiledMap("examples/media/tiled/meadow.tmx")
goal = FindTiledObject(map, "Goal")
If goal = 0 Then RuntimeError "This level has no Goal"
Print TiledObjectType(goal)           ; goal
Print TiledObjectWidth(goal) + " x " + TiledObjectHeight(goal)   ; 32.0 x 96.0

=== TiledObjectPropertyFloat
; a coin carries a value typed in Tiled; use 1.5 when a property is missing

coin = TiledObject(map, 1, "coin")
Print TiledObjectPropertyFloat(coin, "value", 1.5)    ; 10.0

=== TiledObjectHasProperty

=== LoadSheetImage
; a character from Aseprite: trimmed frames come back the same size
Graphics 800, 480, 0, 2
hero = LoadSheetImage("examples/media/tiled/hero.json")
MidHandle hero
DrawImage hero, 100, 100, 0

=== CountSheetTags
; how many named animations the sheet has
hero = LoadSheetImage("examples/media/tiled/hero.json")
Print CountSheetTags(hero)

=== SheetTagName
; list the named animations
hero = LoadSheetImage("examples/media/tiled/hero.json")
For i = 1 To CountSheetTags(hero)
    Print SheetTagName(hero, i)
Next

=== SheetTagFirst
; play the "run" tag of a sprite sheet: its first and last frame set the animation
hero = LoadSheetImage("examples/media/tiled/hero.json")
startFrame = SheetTagFirst(hero, "run")
endFrame = SheetTagLast(hero, "run")
run = CreateAnim2D(hero, startFrame, endFrame, 12)
Print "Run uses frames " + startFrame + " to " + endFrame

=== SheetTagLast
; how long a tag is
hero = LoadSheetImage("examples/media/tiled/hero.json")
Print SheetTagLast(hero, "run") - SheetTagFirst(hero, "run") + 1

=== SheetFrameTime
; how many milliseconds a frame shows
hero = LoadSheetImage("examples/media/tiled/hero.json")
Print SheetFrameTime(hero, 0)

=== Anim2DPlayTag
; safe to ask every frame: it only restarts when the tag changes
hero = LoadSheetImage("examples/media/tiled/hero.json")
anim = CreateAnim2D(hero)
Anim2DPlayTag anim, "run"
DrawImage hero, 100, 100, Anim2DFrame(anim)

=== Anim2DTag
; which tag is playing
hero = LoadSheetImage("examples/media/tiled/hero.json")
anim = CreateAnim2D(hero)
Anim2DPlayTag anim, "idle"
Print Anim2DTag(anim)

=== TiledMapBackground

Cls

=== TiledTileType
; what kind of tile is at a cell

t = GetTile(props, 8, 12)
If t >= 0 Then Print TiledTileType(map, t)   ; hazard

=== TiledTileProperty
