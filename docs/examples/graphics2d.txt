; Examples for the 2D graphics commands. Format: see tools/check_doc_examples.py.

=== Graphics3D
; Open a 800 by 600 window that can draw 3D and 2D
Graphics3D 800, 600, 32, 2   ; mode 2 = window, 1 = fullscreen, 3 = scaled window
camera = CreateCamera()
RenderWorld
Flip

=== Graphics
; Open a 640 by 480 window for 2D drawing only
Graphics 640, 480, 32, 2   ; mode 2 = window
Text 10, 10, "Hello"
Flip

=== EndGraphics
; Close down graphics when the program is finished
Graphics 640, 480
Text 10, 10, "Bye"
Flip
EndGraphics

=== AppTitle
; Set the title shown on the window
Graphics 640, 480
AppTitle "My First Game"

=== SetFullscreen
; Switch between a window and fullscreen while the program runs
Graphics 640, 480
SetFullscreen 1   ; 1 = borderless fullscreen
SetFullscreen 0   ; 0 = back to a window

=== IsFullscreen
; Find out if the program is running fullscreen
Graphics 640, 480
If IsFullscreen() Then Print "Fullscreen" Else Print "In a window"

=== UpdateWorld
; The main loop of a 3D game
Graphics3D 800, 600
camera = CreateCamera()
cube = CreateCube()
MoveEntity cube, 0, 0, 5
While Not KeyHit(1)   ; 1 = Escape
    TurnEntity cube, 0, 1, 0
    UpdateWorld   ; moves animations, collisions and physics on by one frame
    RenderWorld
    Flip
Wend

=== RenderWorld
; Draw the 3D world, then 2D text on top
Graphics3D 800, 600
camera = CreateCamera()
cube = CreateCube()
MoveEntity cube, 0, 0, 5
RenderWorld   ; tween 1 (the default) draws the current positions
Text 10, 10, "Score: 0"
Flip

=== Flip
; Show the finished frame on screen
Graphics 640, 480
While Not KeyHit(1)
    Cls
    Text 10, 10, "Press Escape to quit"
    Flip   ; waits for the display, so the loop runs at screen speed
Wend

=== WireFrame
; Toggle wireframe drawing with the space bar
Graphics3D 800, 600
camera = CreateCamera()
cube = CreateCube()
MoveEntity cube, 0, 0, 5
showWires = 0
While Not KeyHit(1)
    If KeyHit(57) Then showWires = 1 - showWires   ; 57 = space
    WireFrame showWires   ; 1 = wireframe, 0 = solid
    TurnEntity cube, 0, 1, 0
    RenderWorld
    Flip
Wend

=== LoadFont

=== SaveImage

=== DrawImageEx

=== ImagesOverlap

=== LoadImageAsync
; Start loading an image in the background, and wait for it
Graphics 640, 480
background = LoadImageAsync("checker.png")
While ImageLoaded(background) = 0
    Cls
    Text 10, 10, "Loading..."
    Flip
Wend
DrawImage background, 0, 0
Flip

=== ImageLoaded
; 1 = ready, 0 = still loading, -1 = failed
Graphics 640, 480
picture = LoadImageAsync("tree.png")
state = ImageLoaded(picture)
If state = 1 Then Print "Ready"
If state = 0 Then Print "Still loading"
If state = -1 Then Print "Failed"

=== GetColor
; Read a pixel of an image into the drawing colour
Graphics 640, 480

=== LockBuffer
; Lock a buffer while writing many pixels

For x = 0 To 99
    WritePixelFast x, 50, $FFFF0000
Next

Flip

=== UnlockBuffer
; Unlock the buffer when you have finished with fast pixels

Flip

=== ReadPixel
; Read the colour of a pixel and split it into red, green and blue
Graphics 640, 480
picture = CreateImage(8, 8)
SetBuffer ImageBuffer(picture)
WritePixel 1, 1, $FF3366CC
argb = ReadPixel(1, 1)
red = (argb Shr 16) And 255
green = (argb Shr 8) And 255
blue = argb And 255
Print red + ", " + green + ", " + blue
SetBuffer BackBuffer()

=== ReadPixelFast
; Read pixels of a locked buffer
Graphics 640, 480

LockBuffer buffer
WritePixelFast 2, 2, $FFFFFFFF, buffer
argb = ReadPixelFast(2, 2, buffer)
UnlockBuffer buffer
Print "Pixel: " + Hex(argb)
