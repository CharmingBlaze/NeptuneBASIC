; Examples for the 3D world commands. Format: see tools/check_doc_examples.py.

=== EntityClass
; Ask an entity what kind of thing it is
camera = CreateCamera()
box = CreateCube()
Print EntityClass(camera)   ; Camera
Print EntityClass(box)      ; Mesh

=== EntityAutoFade
; The crate fades out between 20 and 50 units away from the camera
camera = CreateCamera()
crate = CreateCube()
EntityAutoFade crate, 20, 50
; EntityAutoFade crate, 0, 0 would turn fading off again

=== ClearWorld
; Throw away the whole level before loading the next one
level = CreateSphere()
tex = CreateTexture(64, 64)
ClearWorld                ; frees entities, brushes and textures
ClearWorld 1, 0, 0        ; or free only the entities

=== CreatePivot
; A pivot is an invisible point; things parented to it move and turn with it
hinge = CreatePivot()
door = CreateCube(hinge)
PositionEntity door, 1, 0, 0
TurnEntity hinge, 0, 90, 0     ; the door swings around the hinge

=== CreateCube
; a crate on the floor: the cube is 2 units wide, so lift it 1 unit to sit on y = 0
Graphics3D 800, 600
floor = CreatePlane()
crate = CreateCube()
PositionEntity crate, 0, 1, 0

=== CreateSphere

marble = CreateSphere(24)
ScaleEntity marble, 0.2, 0.2, 0.2
PositionEntity marble, 0, 0.2, 3
EntityColor marble, 60, 120, 255

=== EntityScaleZ
; With global = 1 the parent's scale is included
parent = CreatePivot()
box = CreateCube(parent)
ScaleEntity parent, 1, 1, 2
ScaleEntity box, 1, 1, 3
Print EntityScaleZ(box)       ; 3.0 (its own scale)
Print EntityScaleZ(box, 1)    ; 6.0 (world scale)

=== CopyEntity
; Make a forest from one tree
tree = CreateCone()
HideEntity tree
For i = 1 To 5
    copy = CopyEntity(tree)
    ShowEntity copy
    PositionEntity copy, i * 3, 0, 10
Next

=== FreeEntity
; Remove a bullet once it has travelled far enough
bullet = CreateSphere()
PositionEntity bullet, 0, 0, 200
If EntityZ(bullet) > 100 Then FreeEntity bullet : bullet = 0

=== EntityParent
; Put a hat on a head and keep it where it is
head = CreateSphere()
hat = CreateCone()
PositionEntity hat, 0, 1, 0
EntityParent hat, head          ; hat keeps its world position
EntityParent hat, 0             ; detach it again

=== GetParent
; Find out what a part is attached to
car = CreatePivot()
wheel = CreateCylinder(8, 1, car)
If GetParent(wheel) = car Then Print "The wheel belongs to the car"

=== CountChildren
; Count the wheels on a car
car = CreatePivot()
For i = 1 To 4
    wheel = CreateCylinder(8, 1, car)
Next
Print CountChildren(car)        ; 4

=== GetChild
; Go through every child of an entity (numbered from 1)
car = CreatePivot()
For i = 1 To 4
    wheel = CreateCylinder(8, 1, car)
Next
For i = 1 To CountChildren(car)
    wheel = GetChild(car, i)
    TurnEntity wheel, 0, 0, 10
Next

=== FindChild
; Look for a named part anywhere inside a model
robot = CreatePivot()
arm = CreateCube(robot)
NameEntity arm, "left_arm"
found = FindChild(robot, "left_arm")
If found <> 0 Then Print "Found the arm"

=== HideEntity

=== EntityHidden

=== PositionEntity
; A child is positioned relative to its parent, unless global is True
Graphics3D 640, 480
tower = CreateCube()
ball = CreateSphere(8, tower)
PositionEntity tower, 10, 0, 0
PositionEntity ball, 0, 5, 0        ; 5 above the tower
DebugLog EntityX(ball, True)        ; 10.0 (where it is in the world)
PositionEntity ball, 0, 5, 0, True  ; now 5 above the world origin instead
DebugLog EntityX(ball, True)        ; 0.0

=== TurnEntity

=== TFormPoint
; Where is the point 1 unit in front of the ship, in world space?
ship = CreateCube()
PositionEntity ship, 10, 0, 0
RotateEntity ship, 0, 90, 0
TFormPoint 0, 0, 1, ship, 0       ; from ship space to world space (0)
Print TFormedX() + ", " + TFormedY() + ", " + TFormedZ()

=== TFormVector
; Which world direction is "forward" for the ship?

=== TFormNormal

=== VectorYaw
; turn an enemy to face the direction it is walking

Print "Straight ahead: " + VectorYaw(0, 0, 1)   ; 0.0

=== VectorPitch

Print "Level ground: " + VectorPitch(0, 0, 1)   ; 0.0

=== TextureFilter
; Give every texture with "checker" in its file name mipmaps (flag 8) automatically.
; Set the rule BEFORE loading: textures that are already loaded keep their flags.
TextureFilter "checker", 1 + 8
floor = LoadTexture("examples/media/checker.png")   ; matches, so it gets 1 + 8
coin = LoadTexture("examples/media/coin.png")       ; no match, so only the default flag
Print TextureName(floor)   ; examples/media/checker.png

=== ClearTextureFilters
; Only the ground textures should be mipmapped: add the rule, load them, then clear it
TextureFilter "checker", 1 + 8
ground = LoadTexture("examples/media/checker.png")   ; mipmapped by the rule
ClearTextureFilters
sign = LoadTexture("examples/media/checker.png")     ; the rule is gone: plain flags
Print TextureName(sign)   ; examples/media/checker.png

=== SetCubeMode

sky = CreateTexture(32, 32, 1 + 128)
For face = 0 To 5
    SetCubeFace sky, face
    SetBuffer TextureBuffer(sky)
    Color 80, 140 + face * 20, 255
    Rect 0, 0, 32, 32
Next
SetBuffer BackBuffer()
SetCubeMode sky, 1   ; 1 reflection (default), 2 looked up by the surface normal, 3 refraction

=== PositionTexture
; Scroll a texture across a floor, like a conveyor belt. The offset is SET each time
; (not added), so give it a value that grows: 1.0 is one whole repeat.
Graphics3D 640, 480
cam = CreateCamera()
PositionEntity cam, 0, 2, -4
RotateEntity cam, 25, 0, 0
belt = CreateCube()
ScaleEntity belt, 2, 0.1, 4
tex = LoadTexture("examples/media/checker.png")
EntityTexture belt, tex
While Not KeyHit(KEY_ESCAPE)
    PositionTexture tex, 0, MilliSecs() / 4000.0   ; one repeat every 4 seconds
    RenderWorld
    Flip
Wend

=== RotateTexture
; Spin the pattern on a cube face. The angle is SET each time (not added), in degrees,
; so keep a running angle yourself.

    angle# = angle# + 0.5

=== EntityShininess

While Not KeyHit(KEY_ESCAPE)

Graphics3D 1280, 720
room = CreateCube()
ScaleEntity room, 5, 3, 5
FlipMesh room                         ; seen from inside
lamp = CreateLight(LIGHT_POINT)
AmbientLight 120, 120, 130            ; the sky light that gets baked
count = BakeLightmaps(4, 1, True)     ; 4 texels a unit, normal quality, lamp light too
HideEntity lamp                       ; its light is baked in now
AmbientLight 20, 20, 25               ; turn down, so the baked light shows
Print count + " surfaces baked"

=== ClearLightmaps
; remove the baked light, for example before baking again after a change
Graphics3D 1280, 720
room = CreateCube()
BakeLightmaps
ClearLightmaps room        ; just this entity and its children
ClearLightmaps             ; or everything

=== EntityHeightMap
; Make a brick wall look deeper with parallax mapping
wall = CreateCube()
heights = LoadTexture("checker.png")
EntityHeightMap wall, heights, 0.1      ; depth scale

=== CreateBrush
; A reusable red brush
red = CreateBrush(255, 0, 0)
box = CreateCube()
PaintEntity box, red

=== LoadBrush
; A brush that already has a texture, tiled 4 times
brick = LoadBrush("checker.png", 1, 4, 4)
wall = CreateCube()
PaintEntity wall, brick

=== FreeBrush
; Free a brush after painting; the paint stays
brush = CreateBrush(0, 200, 0)
box = CreateCube()
PaintEntity box, brush
FreeBrush brush

=== BrushColor
; an orange brush, then paint a crate with it
Graphics3D 800, 600
brush = CreateBrush()
BrushColor brush, 255, 128, 0
crate = CreateCube()
PaintEntity crate, brush

=== BrushAlpha
; a half see-through brush for a ghost
Graphics3D 800, 600
ghostly = CreateBrush()
BrushAlpha ghostly, 0.5
ghost = CreateSphere(16)
PaintEntity ghost, ghostly

=== BrushShininess
; a shiny brush for a polished red ball
Graphics3D 800, 600
shiny = CreateBrush(200, 30, 30)
BrushShininess shiny, 0.7
ball = CreateSphere(24)
PaintEntity ball, shiny

=== BrushTexture
; Put a texture on a brush
brush = CreateBrush()
tex = LoadTexture("checker.png")
BrushTexture brush, tex            ; frame 0, layer 0

=== BrushBlend
; add the brush colour to what is behind it, for a glowing beam
Graphics3D 800, 600
glow = CreateBrush(80, 160, 255)
BrushBlend glow, 3
beam = CreateCube()
PaintEntity beam, glow

=== BrushFX
; full-bright (1) plus no back-face culling (16) for a sign visible from both sides
Graphics3D 800, 600
sign = CreateBrush(255, 255, 255)
BrushFX sign, 1 + 16
board = CreateCube()
PaintEntity board, sign

=== PaintSurface
; Paint just one surface of a mesh
mesh = CreateMesh()
surf = CreateSurface(mesh)
a = AddVertex(surf, 0, 0, 0)
b = AddVertex(surf, 1, 0, 0)
c = AddVertex(surf, 0, 1, 0)
AddTriangle surf, a, b, c
brush = CreateBrush(255, 0, 0)
PaintSurface surf, brush

=== PaintEntity
; Paint a whole cube blue
brush = CreateBrush(0, 0, 255)
box = CreateCube()
PaintEntity box, brush

=== PaintMesh
; Paint every surface of a mesh green
brush = CreateBrush(0, 255, 0)
mesh = CreateMesh()
surf = CreateSurface(mesh)
PaintMesh mesh, brush

=== GetSurfaceBrush
; Copy the look of a surface, then free the copy
mesh = CreateMesh()
surf = CreateSurface(mesh)
copy = GetSurfaceBrush(surf)
FreeBrush copy

=== GetEntityBrush
; Give a second cube the same look as the first
original = CreateCube()
EntityColor original, 255, 0, 0
look = GetEntityBrush(original)
twin = CreateCube()
PaintEntity twin, look
FreeBrush look

=== GetBrushTexture
; Find the texture on a brush
brush = LoadBrush("checker.png")
tex = GetBrushTexture(brush)
If tex <> 0 Then Print TextureName(tex)

=== SkyBox
; Show a cube map as the sky
sky = LoadTexture("sky_cube.png", 1 + 128)
SkyBox sky
SkyBox 0      ; remove the sky
