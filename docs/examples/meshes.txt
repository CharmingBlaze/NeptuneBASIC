; Examples for the mesh, model and collision commands. Format: see tools/check_doc_examples.py.

=== CreateMesh
; An empty mesh; add a surface and vertices to give it a shape
Graphics3D 800, 600
mesh = CreateMesh()
surf = CreateSurface(mesh)
Print "Surfaces: " + CountSurfaces(mesh)

=== CreateSurface
; Surfaces hold the triangles; here the surface is painted red
Graphics3D 800, 600
red = CreateBrush(255, 0, 0)
mesh = CreateMesh()
surf = CreateSurface(mesh, red)
Print "Created surface " + surf

=== CountSurfaces
Graphics3D 800, 600
mesh = CreateMesh()
a = CreateSurface(mesh)
b = CreateSurface(mesh)
Print "Surfaces: " + CountSurfaces(mesh)   ; 2

=== GetSurface
; Surfaces are numbered from 1
Graphics3D 800, 600
mesh = CreateCube()
For i = 1 To CountSurfaces(mesh)
    surf = GetSurface(mesh, i)
    Print "Surface " + i + " has " + CountVertices(surf) + " vertices"
Next

=== AddVertex
; Three corners of a triangle; the returned numbers start at 0
Graphics3D 800, 600
mesh = CreateMesh()
surf = CreateSurface(mesh)
v0 = AddVertex(surf, -1, 0, 0, 0, 1)   ; x, y, z, then texture u, v
v1 = AddVertex(surf, 0, 2, 0, 0.5, 0)
v2 = AddVertex(surf, 1, 0, 0, 1, 1)
AddTriangle surf, v0, v1, v2

=== AddTriangle
; Join three vertices into a triangle and print its number
Graphics3D 800, 600
mesh = CreateMesh()
surf = CreateSurface(mesh)
a = AddVertex(surf, 0, 0, 0)
b = AddVertex(surf, 0, 1, 0)
c = AddVertex(surf, 1, 0, 0)
tri = AddTriangle(surf, a, b, c)
Print "Triangle number " + tri

=== AddAnimSeq

=== SetAnimKey

=== LoadAnimSeq

=== LoadMesh
; Load a model as one entity; 0 means it could not be loaded
Graphics3D 800, 600

=== LoadMeshAsync
; Start loading and carry on; the model appears when it is ready
Graphics3D 800, 600

=== EntityLoaded
; Wait until a background load has finished (-1 means it failed)
Graphics3D 800, 600
model = LoadMeshAsync("examples/media/Fox.glb")
While EntityLoaded(model) = 0
    Delay 10
Wend
If EntityLoaded(model) = -1 Then Print "Load failed" Else Print "Ready"

=== PendingLoads
; Show a loading message while background loads are running
Graphics3D 800, 600
fox = LoadMeshAsync("examples/media/Fox.glb")
man = LoadMeshAsync("examples/media/CesiumMan.glb")
While PendingLoads() > 0
    Print "Still loading: " + PendingLoads()
    Delay 100
Wend

=== EntityPick
; Shoot straight ahead from the player, up to 50 units
Graphics3D 800, 600
player = CreateCube()

PositionEntity target, 0, 0, 10
EntityPickMode target, 1
hit = EntityPick(player, 50)
If hit = target Then Print "Target in the sights"

=== EntityVisible
; Can the guard see the player, or is a wall in the way?

PositionEntity player, 0, 0, 20
wall = CreateCube()
PositionEntity wall, 0, 0, 10
EntityPickMode wall, 2, 1   ; obscurer = 1
If EntityVisible(guard, player) Then Print "Spotted!" Else Print "Hidden"
