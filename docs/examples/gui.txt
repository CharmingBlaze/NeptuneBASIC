; Examples for the GUI widget commands. Format: see tools/check_doc_examples.py.

=== GuiWindow
; A draggable window; widgets created with it as parent go inside
Graphics3D 800, 600, 0, 2
settings = GuiWindow("Settings", 20, 20, 300, 200)   ; title, x, y, width, height
GuiLabel "Welcome!", settings
While GuiValue(settings) = 1 And Not KeyHit(1)       ; 1 while open, 0 once closed
    Flip
Wend

=== GuiPanel
; A row panel holds two buttons side by side
Graphics3D 800, 600, 0, 2
win = GuiWindow("Panel", 20, 20, 300, 100)
row = GuiPanel(1, win)              ; layout 1 = in a row (0 column, 2 wrapping, 3 free)
okButton = GuiButton("OK", row)
cancelButton = GuiButton("Cancel", row)

=== GuiGrid
; A 2 by 2 grid with a label and a text box on the first row
Graphics3D 800, 600, 0, 2
win = GuiWindow("Grid", 20, 20, 300, 120)
grid = GuiGrid(2, 2, win)           ; 2 columns, 2 rows
nameLabel = GuiLabel("Name:", grid)
nameBox = GuiTextBox("Bob", grid)
GuiCell nameLabel, 0, 0
GuiCell nameBox, 1, 0

=== GuiCell
; Put a label in the second column of the first row of a grid
Graphics3D 800, 600, 0, 2
grid = GuiGrid(3, 2)
title = GuiLabel("Score", grid)
GuiCell title, 1, 0                 ; column 1, row 0 (counting from 0)

=== GuiLabel
; A label whose text changes with the score
Graphics3D 800, 600, 0, 2

While Not KeyHit(1)
    If KeyHit(57) Then score = score + 10 : SetGuiText scoreLabel, "Score: " + score   ; space bar
    Flip
Wend

=== SetGuiValue
; Set a slider, a progress bar and a check box from the program

=== GuiValue

=== GuiClicked
; Count button presses since the last check
Graphics3D 800, 600, 0, 2
jump = GuiButton("Jump")
presses = 0
While Not KeyHit(1)
    presses = presses + GuiClicked(jump)
    Flip
Wend
Print "Pressed " + presses + " times"

=== GuiUnderMouse
; Only fire when the click was not meant for the GUI
Graphics3D 800, 600, 0, 2
menuButton = GuiButton("Menu")
While Not KeyHit(1)
    If MouseHit(1) And GuiUnderMouse() = 0 Then Print "Shot fired"
    Flip
Wend

=== GuiTreeItem
; Build a small tree: World with Player and Enemies inside

While Not KeyHit(1)
    If GuiClicked(player) > 0 Then Print "Player selected"
    Flip
Wend

=== GuiMessageBox
; Ask a yes/no question

    If GuiChanged(box) > 0 Then
        If GuiValue(box) = 1 Then Print "Yes" Else Print "No"   ; 1 = Yes, 2 = No
        Exit
    EndIf
    Flip
Wend
