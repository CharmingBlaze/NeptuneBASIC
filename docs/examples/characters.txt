; Examples for character controllers. Format: see tools/check_doc_examples.py.

=== CreateCharController
; a hero whose entity origin is at its feet, walking on a static floor
Graphics3D 800, 600
floor = CreateCube()
ScaleEntity floor, 20, 0.5, 20
PositionEntity floor, 0, -0.5, 0
EntityPhysics floor, 2                ; body_type 2 = static: solid, never moves
hero = CreateCube()
PositionEntity hero, 0, 1, 0
player = CreateCharController(hero, 0.4, 1.8)   ; radius 0.4, height 1.8
cam = CreateCamera()
PositionEntity cam, 0, 3, -8
While Not KeyHit(KEY_ESCAPE)
    CharMove player, KeyDown(KEY_D) - KeyDown(KEY_A), 0, KeyDown(KEY_W) - KeyDown(KEY_S)
    UpdateChar player
    UpdateWorld
    RenderWorld
    Flip
Wend

=== CreateCharController2D
; a platformer hero: a 26 by 44 box with its feet at 100, 400, falling onto the floor
Graphics 960, 540
floor = CreateBody2D(1, 480, 520)     ; body type 1 = static
Body2DBox floor, 960, 40
player = CreateCharController2D(100, 400, 26, 44)
While Not KeyHit(KEY_ESCAPE)
    CharMove player, KeyDown(KEY_RIGHT) - KeyDown(KEY_LEFT)
    UpdateChar player
    Cls
    Rect CharX(player) - 13, CharY(player) - CharHeight(player), 26, CharHeight(player)
    Flip
Wend

=== FreeCharController
; the entity stays, but no longer walks by itself
hero = CreatePivot()
player = CreateCharController(hero)
FreeCharController player

=== UpdateChar
; once a frame, after giving it input
hero = CreatePivot()
player = CreateCharController(hero)
UpdateChar player

=== CharMove
; run along +X
hero = CreatePivot()
player = CreateCharController(hero)
CharMove player, 1, 0, 0

=== CharWalk
; W means away from the camera, whichever way it faces
hero = CreatePivot()
player = CreateCharController(hero)
camera = CreateCamera()
CharWalk player, KeyDown(KEY_W) - KeyDown(KEY_S), KeyDown(KEY_D) - KeyDown(KEY_A), EntityYaw(camera)

=== CharJump

=== CharDrop
; a thin platform the player can jump onto, and drop through with Down + Space
Graphics 960, 540
ground = CreateBody2D(1, 480, 520)
Body2DBox ground, 960, 40
ledge = CreateBody2D(1, 480, 400)
Body2DBox ledge, 200, 12
Body2DOneWay ledge                    ; solid from above, passable from below
player = CreateCharController2D(480, 300)
While Not KeyHit(KEY_ESCAPE)
    If KeyHit(KEY_SPACE)
        If KeyDown(KEY_DOWN) Then CharDrop player Else CharJump player
    EndIf
    UpdateChar player
    Cls
    Rect 380, 394, 200, 12
    Rect CharX(player) - 12, CharY(player) - 32, 24, 32
    Flip
Wend

=== CharMode
; a top-down game: no gravity, so CharMove can go in any direction, and walls still stop it
Graphics 960, 540
player = CreateCharController2D(480, 270, 24, 24)
CharMode player, CHAR_FLY
While Not KeyHit(KEY_ESCAPE)
    CharMove player, KeyDown(KEY_RIGHT) - KeyDown(KEY_LEFT), KeyDown(KEY_DOWN) - KeyDown(KEY_UP)
    UpdateChar player
    Cls
    Rect CharX(player) - 12, CharY(player) - 24, 24, 24
    Flip
Wend

=== CharHeight
; draw a 2D character at its current height: it gets shorter while Down is held
Graphics 960, 540
floor = CreateBody2D(1, 480, 520)
Body2DBox floor, 960, 40
player = CreateCharController2D(100, 400, 26, 44)
While Not KeyHit(KEY_ESCAPE)
    CharCrouch player, KeyDown(KEY_DOWN)
    UpdateChar player
    Cls
    Rect CharX(player) - 13, CharY(player) - CharHeight(player), 26, CharHeight(player)
    Flip
Wend
