; Examples for the 3D world commands. Format: see tools/check_doc_examples.py.

=== EntityClass
; Ask an entity what kind of thing it is
camera = CreateCamera()
box = CreateCube()
Print EntityClass(camera)   ; Camera
Print EntityClass(box)      ; Mesh

=== EntityAutoFade
; The crate fades out between 20 and 50 units away from the camera
camera = CreateCamera()
crate = CreateCube()
EntityAutoFade crate, 20, 50
; EntityAutoFade crate, 0, 0 would turn fading off again

=== ClearWorld
; Throw away the whole level before loading the next one
level = CreateSphere()
tex = CreateTexture(64, 64)
ClearWorld                ; frees entities, brushes and textures
ClearWorld 1, 0, 0        ; or free only the entities

=== CreatePivot
; A pivot is an invisible point; things parented to it move and turn with it
hinge = CreatePivot()
door = CreateCube(hinge)
PositionEntity door, 1, 0, 0
TurnEntity hinge, 0, 90, 0     ; the door swings around the hinge

=== CreateCube
; Make a cube and move it up
box = CreateCube()
PositionEntity box, 0, 1, 5

=== CreateSphere
; A smoother ball: more segments means more detail
ball = CreateSphere(16)
PositionEntity ball, 0, 1, 5

=== CreateCylinder
; A pillar with 12 sides, and a pipe with open ends
pillar = CreateCylinder(12)
pipe = CreateCylinder(12, 0)     ; solid = 0 leaves the ends open
PositionEntity pillar, -3, 0, 8
PositionEntity pipe, 3, 0, 8

=== CreateCone
; A tree-top cone with 16 sides
treetop = CreateCone(16)
PositionEntity treetop, 0, 2, 6
ScaleEntity treetop, 1, 1.5, 1

=== CreatePlane
; A flat floor that goes on forever
floor = CreatePlane()
EntityColor floor, 80, 160, 80

=== EntityScaleX
; Read back the width of a stretched box
box = CreateCube()
ScaleEntity box, 3, 1, 1
Print EntityScaleX(box)       ; 3.0

=== EntityScaleY
; Read back the height of a stretched box
box = CreateCube()
ScaleEntity box, 1, 2, 1
Print EntityScaleY(box)       ; 2.0

=== EntityScaleZ
; With global = 1 the parent's scale is included
parent = CreatePivot()
box = CreateCube(parent)
ScaleEntity parent, 1, 1, 2
ScaleEntity box, 1, 1, 3
Print EntityScaleZ(box)       ; 3.0 (its own scale)
Print EntityScaleZ(box, 1)    ; 6.0 (world scale)

=== CopyEntity
; Make a forest from one tree
tree = CreateCone()
HideEntity tree
For i = 1 To 5
    copy = CopyEntity(tree)
    ShowEntity copy
    PositionEntity copy, i * 3, 0, 10
Next

=== FreeEntity
; Remove a bullet once it has travelled far enough
bullet = CreateSphere()
PositionEntity bullet, 0, 0, 200
If EntityZ(bullet) > 100 Then FreeEntity bullet : bullet = 0

=== EntityParent
; Put a hat on a head and keep it where it is
head = CreateSphere()
hat = CreateCone()
PositionEntity hat, 0, 1, 0
EntityParent hat, head          ; hat keeps its world position
EntityParent hat, 0             ; detach it again

=== GetParent
; Find out what a part is attached to
car = CreatePivot()
wheel = CreateCylinder(8, 1, car)
If GetParent(wheel) = car Then Print "The wheel belongs to the car"

=== CountChildren
; Count the wheels on a car
car = CreatePivot()
For i = 1 To 4
    wheel = CreateCylinder(8, 1, car)
Next
Print CountChildren(car)        ; 4

=== GetChild
; Go through every child of an entity (numbered from 1)
car = CreatePivot()
For i = 1 To 4
    wheel = CreateCylinder(8, 1, car)
Next
For i = 1 To CountChildren(car)
    wheel = GetChild(car, i)
    TurnEntity wheel, 0, 0, 10
Next

=== FindChild
; Look for a named part anywhere inside a model
robot = CreatePivot()
arm = CreateCube(robot)
NameEntity arm, "left_arm"
found = FindChild(robot, "left_arm")
If found <> 0 Then Print "Found the arm"

=== NameEntity
; Give an entity a name so you can find it later
door = CreateCube()
NameEntity door, "front_door"
Print EntityName(door)

=== EntityName
; Print the name of an entity
door = CreateCube()
NameEntity door, "front_door"
Print "This is the " + EntityName(door)

=== HideEntity
; Hide an enemy while it is dead
enemy = CreateCube()
HideEntity enemy

=== ShowEntity
; Bring a hidden enemy back
enemy = CreateCube()
HideEntity enemy
ShowEntity enemy

=== EntityHidden
; Only update an enemy while it is visible
enemy = CreateCube()
HideEntity enemy
If EntityHidden(enemy) = 1 Then Print "The enemy is hidden"

=== PositionEntity
; Place a ball 5 units up and 10 units in front of the origin
ball = CreateSphere()
PositionEntity ball, 0, 5, 10

=== MoveEntity
; Fly a ship forward the way it faces
ship = CreateCube()
RotateEntity ship, 0, 90, 0
MoveEntity ship, 0, 0, 0.5       ; z is forward

=== TranslateEntity
; Push a crate along the world X axis, whichever way it faces
crate = CreateCube()
RotateEntity crate, 0, 45, 0
TranslateEntity crate, 1, 0, 0

=== RotateEntity
; Face the entity 90 degrees to the side
statue = CreateCone()
RotateEntity statue, 0, 90, 0     ; pitch, yaw, roll

=== TurnEntity
; Spin a coin a little each frame
coin = CreateCylinder()
TurnEntity coin, 0, 3, 0

=== ScaleEntity
; Make a flat wide platform
platform = CreateCube()
ScaleEntity platform, 4, 0.25, 4

=== PointEntity
; Make a turret track the player

PositionEntity player, 10, 0, 10
PointEntity turret, player

=== LookAt

LookAt lamp, 5, 0, 5

=== CameraLookAt
; Point the camera at the middle of the scene
camera = CreateCamera()

=== EntityX
; Check where a ball is along X
ball = CreateSphere()
PositionEntity ball, 4, 0, 0
Print EntityX(ball)

=== EntityY
; Stop a falling ball at the ground
ball = CreateSphere()
PositionEntity ball, 0, -3, 0
If EntityY(ball) < 0 Then PositionEntity ball, EntityX(ball), 0, EntityZ(ball)
Print EntityY(ball)

=== EntityZ
; Child positions are relative to the parent unless global is 1
parent = CreatePivot()
PositionEntity parent, 0, 0, 10
child = CreateCube(parent)
PositionEntity child, 0, 0, 2
Print EntityZ(child)        ; 2.0 (local)
Print EntityZ(child, 1)     ; 12.0 (world)

=== EntityPitch
; Read how far a cannon is tilted

Print EntityPitch(cannon)

=== EntityYaw
; Read which way a ship is turned
ship = CreateCube()
RotateEntity ship, 0, 45, 0
Print EntityYaw(ship)

=== EntityRoll
; Read how far a plane is banked
plane = CreateCube()
RotateEntity plane, 0, 0, 20
Print EntityRoll(plane)

=== TFormPoint
; Where is the point 1 unit in front of the ship, in world space?
ship = CreateCube()
PositionEntity ship, 10, 0, 0
RotateEntity ship, 0, 90, 0
TFormPoint 0, 0, 1, ship, 0       ; from ship space to world space (0)
Print TFormedX() + ", " + TFormedY() + ", " + TFormedZ()

=== TFormVector
; Which world direction is "forward" for the ship?
ship = CreateCube()
RotateEntity ship, 0, 90, 0
TFormVector 0, 0, 1, ship, 0
Print TFormedX() + ", " + TFormedY() + ", " + TFormedZ()

=== TFormNormal
; Same as TFormVector, but the result has length 1
ship = CreateCube()
ScaleEntity ship, 3, 3, 3
RotateEntity ship, 0, 90, 0
TFormNormal 0, 0, 1, ship, 0
Print TFormedX() + ", " + TFormedY() + ", " + TFormedZ()

=== TFormedX
; Read the X of the last TForm result
ship = CreateCube()
PositionEntity ship, 5, 0, 0
TFormPoint 0, 0, 0, ship, 0
Print TFormedX()          ; 5.0

=== TFormedY
; Read the Y of the last TForm result
ship = CreateCube()
PositionEntity ship, 0, 7, 0
TFormPoint 0, 0, 0, ship, 0
Print TFormedY()          ; 7.0

=== TFormedZ
; Read the Z of the last TForm result
ship = CreateCube()
PositionEntity ship, 0, 0, 9
TFormPoint 0, 0, 0, ship, 0
Print TFormedZ()          ; 9.0

=== VectorYaw

=== DeltaYaw
; How far must a guard turn to face the player?
guard = CreateCube()
player = CreateSphere()
PositionEntity player, 10, 0, 10
amount = DeltaYaw(guard, player)
TurnEntity guard, 0, amount, 0

=== DeltaPitch
; How far must a turret tilt to aim at a plane?
turret = CreateCone()
plane = CreateCube()
PositionEntity plane, 0, 10, 10
TurnEntity turret, DeltaPitch(turret, plane), 0, 0

=== AlignToVector
; Tilt a tree so its up axis (2 = Y) follows the slope of the ground
tree = CreateCone()
AlignToVector tree, 0.3, 1, 0, 2        ; axis 1 = X, 2 = Y, 3 = Z
AlignToVector tree, 0, 1, 0.3, 2, 0.1   ; rate 0.1 turns it only a tenth of the way

=== GetMatElement
; Read the "forward" direction straight from the rotation matrix
ship = CreateCube()
RotateEntity ship, 0, 90, 0
forwardx = GetMatElement(ship, 2, 0)    ; row 2, column 0
Print forwardx

=== EntityDistance

=== LoadAnimTexture

=== TextureFilter

=== LoadTextureAsync
; Start loading a texture without freezing the game
tex = LoadTextureAsync("checker.png")
box = CreateCube()
EntityTexture box, tex          ; white until it has loaded

=== TextureLoaded
; Check whether a texture is ready
tex = LoadTextureAsync("checker.png")
If TextureLoaded(tex) = 1 Then Print "Ready"
If TextureLoaded(tex) = -1 Then Print "Loading failed"

=== LoadTexture
; Load a picture and put it on a cube
tex = LoadTexture("checker.png", 1 + 8)    ; colour + mipmapped
If tex = 0 Then RuntimeError "checker.png is missing"
box = CreateCube()
EntityTexture box, tex

=== BakeLightmaps
; bake soft sky light, corner shadows and bounced light into a still room
Graphics3D 1280, 720
room = CreateCube()
ScaleEntity room, 5, 3, 5
FlipMesh room                         ; seen from inside
lamp = CreateLight(LIGHT_POINT)
AmbientLight 120, 120, 130            ; the sky light that gets baked
count = BakeLightmaps(4, 1, True)     ; 4 texels a unit, normal quality, lamp light too
HideEntity lamp                       ; its light is baked in now
AmbientLight 20, 20, 25               ; turn down, so the baked light shows
Print count + " surfaces baked"

=== ClearLightmaps
; remove the baked light, for example before baking again after a change
Graphics3D 1280, 720
room = CreateCube()
BakeLightmaps
ClearLightmaps room        ; just this entity and its children
ClearLightmaps             ; or everything

=== PaintSurface
; Paint just one surface of a mesh
mesh = CreateMesh()
surf = CreateSurface(mesh)
a = AddVertex(surf, 0, 0, 0)
b = AddVertex(surf, 1, 0, 0)
c = AddVertex(surf, 0, 1, 0)
AddTriangle surf, a, b, c
brush = CreateBrush(255, 0, 0)
PaintSurface surf, brush

=== PaintEntity

=== PaintMesh
; Paint every surface of a mesh green

=== GetSurfaceBrush
; Copy the look of a surface, then free the copy
mesh = CreateMesh()
surf = CreateSurface(mesh)
copy = GetSurfaceBrush(surf)
FreeBrush copy

=== GetEntityBrush

PaintEntity twin, look
