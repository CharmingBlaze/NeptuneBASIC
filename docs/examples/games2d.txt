; Examples for the 2D game commands. Format: see tools/check_doc_examples.py.

=== CreateTileMap
Graphics 640, 480
tiles = LoadAnimImage("tiles.png", 32, 32, 0, 16)
; a map 20 tiles wide and 15 tall, drawn with the frames of tiles
map = CreateTileMap(tiles, 20, 15)
Print "Map size: " + TileMapWidth(map) + " x " + TileMapHeight(map)

=== LoadTileMap
Graphics 640, 480
tiles = LoadAnimImage("tiles.png", 32, 32, 0, 16)
; load a map saved earlier; it returns 0 if the file cannot be read
map = LoadTileMap("level1.map", tiles)
If map = 0 Then Print "Could not load level1.map"

=== SaveTileMap
Graphics 640, 480
tiles = LoadAnimImage("tiles.png", 32, 32, 0, 16)
map = CreateTileMap(tiles, 20, 15)
SetTile map, 3, 4, 1
; returns 1 when the file was written, 0 on failure
If SaveTileMap(map, "level1.map") Then Print "Saved" Else Print "Save failed"

=== FreeTileMap
Graphics 640, 480
tiles = LoadAnimImage("tiles.png", 32, 32, 0, 16)
map = CreateTileMap(tiles, 20, 15)
; ... use the map ...
FreeTileMap map                ; finished with it

=== SetTile
Graphics 640, 480
tiles = LoadAnimImage("tiles.png", 32, 32, 0, 16)
map = CreateTileMap(tiles, 20, 15)
; put frame 5 in the cell at column 3, row 4 (counting from 0)
SetTile map, 3, 4, 5

=== GetTile
Graphics 640, 480
tiles = LoadAnimImage("tiles.png", 32, 32, 0, 16)
map = CreateTileMap(tiles, 20, 15)
SetTile map, 3, 4, 5
frame = GetTile(map, 3, 4)     ; the frame number stored in that cell
Print "Tile at 3,4 is frame " + frame

=== TileMapWidth
Graphics 640, 480
tiles = LoadAnimImage("tiles.png", 32, 32, 0, 16)
map = CreateTileMap(tiles, 20, 15)
columns = TileMapWidth(map)    ; width in cells
Print "The map has " + columns + " columns"

=== TileMapHeight
Graphics 640, 480
tiles = LoadAnimImage("tiles.png", 32, 32, 0, 16)
map = CreateTileMap(tiles, 20, 15)
rows = TileMapHeight(map)      ; height in cells

=== TileMapImage
Graphics 640, 480
tiles = LoadAnimImage("tiles.png", 32, 32, 0, 16)
map = CreateTileMap(tiles, 20, 15)
; switch the whole map to a different tile set (for example a winter theme)
snowTiles = LoadAnimImage("tiles_snow.png", 32, 32, 0, 16)
TileMapImage map, snowTiles

=== TileMapTerrain
Graphics 640, 480
tiles = LoadAnimImage("tiles.png", 32, 32, 0, 16)
map = CreateTileMap(tiles, 20, 15)
; terrain 1 uses 16 frames starting at frame 0 (mode 16 = four neighbours)
TileMapTerrain map, 1, 0, 16

=== AutoTile
Graphics 640, 480
tiles = LoadAnimImage("tiles.png", 32, 32, 0, 16)
map = CreateTileMap(tiles, 20, 15)
TileMapTerrain map, 1, 0, 16
; paint terrain 1 along a row of cells; the frames join up with their neighbours
For column = 2 To 8
    AutoTile map, column, 5, 1
Next

=== GetTileTerrain
Graphics 640, 480
tiles = LoadAnimImage("tiles.png", 32, 32, 0, 16)
map = CreateTileMap(tiles, 20, 15)
TileMapTerrain map, 1, 0, 16
AutoTile map, 4, 4, 1
terrain = GetTileTerrain(map, 4, 4)   ; 1 here; an unpainted cell gives 0
Print "Terrain at 4,4: " + terrain

=== TileAt

=== CreateAnim2D
Graphics 640, 480
frames = LoadAnimImage("hero.png", 32, 32, 0, 8)
; frames 0 to 3 at 10 frames a second; mode 1 = loop (2 ping-pong, 3 play once)
walk = CreateAnim2D(frames, 0, 3, 10, 1)

=== Anim2DPlay
Graphics 640, 480
frames = LoadAnimImage("hero.png", 32, 32, 0, 8)
anim = CreateAnim2D(frames, 0, 3, 10, 1)
; switch to frames 4 to 7 at 12 frames a second, playing once (mode 3)
Anim2DPlay anim, 4, 7, 12, 3

=== Anim2DPause
Graphics 640, 480
frames = LoadAnimImage("hero.png", 32, 32, 0, 8)
anim = CreateAnim2D(frames, 0, 3, 10)
Anim2DPause anim, 1            ; 1 = pause
Anim2DPause anim, 0            ; 0 = resume

=== Anim2DFrame
Graphics 640, 480
frames = LoadAnimImage("hero.png", 32, 32, 0, 8)
anim = CreateAnim2D(frames, 0, 3, 10)
current = Anim2DFrame(anim)    ; the frame number showing right now
Print "Showing frame " + current

=== Anim2DPlaying
Graphics 640, 480
frames = LoadAnimImage("hero.png", 32, 32, 0, 8)
anim = CreateAnim2D(frames, 4, 7, 12, 3)   ; plays once
If Anim2DPlaying(anim) = 0 Then Print "Finished" Else Print "Still playing"

=== DrawAnim2D
Graphics 640, 480
frames = LoadAnimImage("hero.png", 32, 32, 0, 8)
anim = CreateAnim2D(frames, 0, 3, 10)
While Not KeyHit(1)
    Cls
    DrawAnim2D anim, 320, 240      ; draws the current frame, like DrawImage
    Flip
Wend

=== FreeAnim2D
Graphics 640, 480
frames = LoadAnimImage("hero.png", 32, 32, 0, 8)
anim = CreateAnim2D(frames, 0, 3, 10)
FreeAnim2D anim                ; frees the animation, not the image
FreeImage frames

=== ContactBody2D
Graphics 640, 480
Physics2DGravity 0, 600
hero = CreateBody2D(0, 100, 300)
Body2DCapsule hero, 20, 40
; go through every body the hero is touching (numbered from 1)
For index = 1 To CountContacts2D(hero)
    other = ContactBody2D(hero, index)
    Print "Touching body " + other
Next

=== DrawMovie
Graphics 640, 480
movie = OpenMovie("intro.gif")
While Not KeyHit(1)
    Cls
    ; draw at 0,0 scaled to 640 x 480; returns 0 once it has finished
    If DrawMovie(movie, 0, 0, 640, 480) = 0 Then Exit
    Flip
Wend
CloseMovie movie
