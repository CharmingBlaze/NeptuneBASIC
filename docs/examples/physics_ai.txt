=== EntityPhysics
; A dynamic box that falls, and a static floor it lands on
floor = CreateCube()
ScaleEntity floor, 10, 0.5, 10
EntityPhysics floor, 2, 1            ; body_type 2 = static, shape 1 = box
crate = CreateCube()
PositionEntity crate, 0, 5, 0
EntityPhysics crate, 1, 1, 2.5       ; dynamic, box shape, 2.5 kg

=== PhysicsGravity

cam = CreateCamera() : PositionEntity cam, 0, 3, -12
ball = CreateSphere() : PositionEntity ball, 0, 8, 0 : EntityPhysics ball, 1, 2, 1
PhysicsGravity 0, -1.6, 0
While Not KeyHit(1)
    UpdateWorld : RenderWorld : Flip
Wend

=== CreatePathGrid
; a 5 x 5 map with a wall across column 2, then a route around it

    SetPathCost grid, 2, row, 0              ; 0 = wall
Next

=== SetPathCost

=== FindGridPath

    SetPathCost grid, 10, row, 0

=== CreateBehavior
; Build a guard's brain: attack if the player is close, otherwise patrol
brain = CreateBehavior("guard")
root = BehaviorSelector(brain)
attack = BehaviorSequence(brain, root)
BehaviorIf brain, "dist < 5", attack
BehaviorDo brain, "attack", attack
BehaviorDo brain, "patrol", root

=== LoadBehavior
; The file has one node per line, children indented under their parent
brain = LoadBehavior("guard.txt")
guard = CreateCube()
SetBehavior guard, brain

=== BehaviorFromText
text = "selector" + Chr(10)
text = text + "  sequence" + Chr(10)
text = text + "    if dist < 5" + Chr(10)
text = text + "    do attack" + Chr(10)
text = text + "  do patrol"
brain = BehaviorFromText(text)

=== BehaviorSelector
; A selector tries its children in order; the first one that applies wins
brain = CreateBehavior()
root = BehaviorSelector(brain)
BehaviorDo brain, "idle", root

=== BehaviorSequence
brain = CreateBehavior()
root = BehaviorSequence(brain)        ; runs its children one after the other
BehaviorDo brain, "walk to door", root
BehaviorDo brain, "open door", root

=== BehaviorInvert
; "Not hungry" is the opposite of "hungry"
brain = CreateBehavior()
root = BehaviorSequence(brain)
notHungry = BehaviorInvert(brain, root)
BehaviorIf brain, "hungry", notHungry
BehaviorDo brain, "play", root

=== BehaviorIf
brain = CreateBehavior()
root = BehaviorSequence(brain)
BehaviorIf brain, "health < 20", root   ; compares a parameter with a number
BehaviorDo brain, "flee", root

=== BehaviorDo
brain = CreateBehavior()
root = BehaviorSelector(brain)
BehaviorDo brain, "wander", root     ; BehaviorAction returns this name

=== FreeBehavior
brain = CreateBehavior()
BehaviorDo brain, "idle"
FreeBehavior brain

=== SetBehavior
brain = CreateBehavior()
BehaviorDo brain, "idle"
guard = CreateCube()
SetBehavior guard, brain             ; each character keeps its own parameters
SetBehavior guard, 0                 ; 0 removes the behaviour

=== SetBehaviorParam
; Tell the brain how far away the player is, every frame
brain = CreateBehavior()
BehaviorDo brain, "idle"
guard = CreateCube()
player = CreateCube()
SetBehavior guard, brain
SetBehaviorParam guard, "dist", EntityDistance(guard, player)

=== BehaviorParam
brain = CreateBehavior()
BehaviorDo brain, "idle"
guard = CreateCube()
SetBehavior guard, brain
SetBehaviorParam guard, "health", 80
Print "Health: " + BehaviorParam(guard, "health")

=== BehaviorAction
; Ask the brain what to do, then do it
text = "selector" + Chr(10) + "  sequence" + Chr(10) + "    if dist < 5" + Chr(10) + "    do attack" + Chr(10) + "  do patrol"
brain = BehaviorFromText(text)
guard = CreateCube()
SetBehavior guard, brain
SetBehaviorParam guard, "dist", 3
action = BehaviorAction(guard)
If action = "attack" Then Print "Attacking!"
If action = "patrol" Then Print "Patrolling"

=== BehaviorResult
brain = CreateBehavior()
BehaviorDo brain, "open door"
guard = CreateCube()
SetBehavior guard, brain
action = BehaviorAction(guard)
If action = "open door" Then BehaviorResult guard, 1   ; 1 = it worked, 0 = it failed

ProbeOffset probe, 0, 1.5, 0         ; take the picture from 1.5 units higher

=== ProbeTexture
; Use the probe's picture in a custom shader
probe = CreateReflectionProbe()
cube = CreateCube()
env = ProbeTexture(probe)
ShaderTexture cube, "envMap", env

=== EnvironmentLighting
; light a shiny ball from its surroundings (a reflection probe) instead of the flat ambient colour
Graphics3D 800, 600
cam = CreateCamera()
ball = CreateSphere(32)
PositionEntity ball, 0, 0, 4
EntityShininess ball, 1
probe = CreateReflectionProbe()
PositionEntity probe, 0, 1, 4
EnvironmentLighting 1, 0.8            ; 1 = sky and probes, a little dimmer than normal
While Not KeyHit(KEY_ESCAPE)
    RenderWorld
    Flip
Wend

emitter = CreateEmitter()

=== EmitterLife
emitter = CreateEmitter()
EmitterLife emitter, 1, 3            ; each particle lives 1 to 3 seconds

=== EmitterVelocity
; a fountain: particles shoot up at 6 units a second with a little sideways spread, then gravity pulls them down
Graphics3D 800, 600
cam = CreateCamera()
PositionEntity cam, 0, 2, -8
fountain = CreateEmitter()
EmitterVelocity fountain, 0, 6, 0, 1
EmitterGravity fountain, 0, -9, 0
While Not KeyHit(KEY_ESCAPE)
    UpdateWorld
    RenderWorld
    Flip
Wend

emitter = CreateEmitter()
EmitterSize emitter, 0.2, 0.5        ; start size 0.2, grow 0.5 per second

=== EmitterSpin
emitter = CreateEmitter()
EmitterSpin emitter, -90, 90         ; each particle spins at a random speed in this range

=== EmitterColor
emitter = CreateEmitter()
EmitterColor emitter, 255, 200, 50, 255   ; yellow, fully visible

=== EmitterFadeColor
; Fire: start yellow and fade to transparent red
emitter = CreateEmitter()
EmitterColor emitter, 255, 200, 50
EmitterFadeColor emitter, 255, 0, 0, 0

=== EmitterGravity
emitter = CreateEmitter()
EmitterGravity emitter, 0, -9.8, 0   ; particles fall like rain
EmitterGravity emitter, 0, 2, 0      ; or float upwards like smoke

=== EmitterShape
emitter = CreateEmitter()
EmitterShape emitter, 0, 1           ; shape 0: a sphere of radius 1
EmitterShape emitter, 1, 4, 0.1, 4   ; shape 1: a flat box 4 wide and 4 deep

=== EmitterSoftness
emitter = CreateEmitter()
EmitterSoftness emitter, 0.5         ; particles fade out where they touch the ground
EmitterSoftness emitter, 0           ; 0 turns it off

=== EmitterLocal
; a carried torch: with local particles the flames move along with it instead of leaving a trail
Graphics3D 800, 600
cam = CreateCamera()
torch = CreateEmitter()
EmitterLocal torch, True              ; try False to see the trail
While Not KeyHit(KEY_ESCAPE)
    PositionEntity torch, Sin(MilliSecs() / 10.0) * 3, 0, 6
    UpdateWorld
    RenderWorld
    Flip
Wend

=== EmitterEnable
emitter = CreateEmitter()
EmitterEnable emitter, 0             ; stop making new particles
EmitterEnable emitter, 1             ; start again

=== EmitterClear
emitter = CreateEmitter()
EmitterClear emitter                 ; every particle disappears at once

=== EntityLOD
; A statue with a detailed model up close and a simple one far away
statue = CreatePivot()
detailed = CreateSphere(32)
simple = CreateCube()
EntityParent detailed, statue, 0
EntityParent simple, statue, 0
EntityLOD statue, detailed, 0, 30    ; shown from 0 to 30 units from the camera
EntityLOD statue, simple, 30, 0      ; shown beyond 30 units (far 0 = no upper limit)

=== CameraToTexture
; Show a security camera's view on a screen in the world
securityCamera = CreateCamera()
tex = CreateTexture(256, 256)
CameraToTexture securityCamera, tex
monitor = CreateCube()
EntityTexture monitor, tex
CameraToTexture securityCamera, 0    ; texture 0 renders to the screen again

=== DebugLine
; show where a ray goes: from the player 20 units forward, in red
player = CreatePivot()
TFormPoint 0, 0, 20, player, 0
DebugLine EntityX(player), EntityY(player), EntityZ(player), TFormedX(), TFormedY(), TFormedZ(), 255, 0, 0

=== DebugBox
; show the area a trigger covers, in green, every frame
Graphics3D 800, 600
cam = CreateCamera() : PositionEntity cam, 0, 3, -12
While Not KeyHit(1)
    DebugBox 0, 1, 0, 4, 2, 4, 0, 255, 0
    RenderWorld : Flip
Wend

=== DebugSphere
; show how far an enemy can hear: an orange sphere of radius 15 that follows it
Graphics3D 800, 600
cam = CreateCamera() : PositionEntity cam, 0, 5, -40
enemy = CreateCube()
While Not KeyHit(1)
    TurnEntity enemy, 0, 1, 0 : MoveEntity enemy, 0, 0, 0.1
    DebugSphere EntityX(enemy), EntityY(enemy), EntityZ(enemy), 15, 255, 160, 0
    RenderWorld : Flip
Wend

=== DebugPhysics
; show the colliders of a falling ball and the floor, until turned off
Graphics3D 800, 600
cam = CreateCamera() : PositionEntity cam, 0, 3, -10
floor = CreateCube() : ScaleEntity floor, 8, 0.5, 8 : EntityPhysics floor, 2, 1
ball = CreateSphere() : PositionEntity ball, 0, 5, 0 : EntityPhysics ball, 1, 2, 1
DebugPhysics True
While Not KeyHit(1)
    UpdateWorld : RenderWorld : Flip
Wend

=== DebugCollisions

player = CreateSphere() : EntityType player, 1 : EntityRadius player, 1
wall = CreateCube() : MoveEntity wall, 4, 0, 0 : EntityType wall, 2 : EntityBox wall, -1, -1, -1, 2, 2, 2
DebugCollisions True
While Not KeyHit(1)
    RenderWorld : Flip
Wend
