; Examples for easing, settings, input actions, tweens, camera shake, screen
; fades, 2D particles and gamepad rumble. Format: see tools/check_doc_examples.py.

=== Ease
; a box that glides in from the left and settles
For frame = 0 To 60
    x# = -200 + 400 * Ease(frame / 60.0, EASE_OUT_CUBIC)
Next

=== EaseBetween
; a score counter that rolls up from 0 to 1500 in one second
shown = EaseBetween(0, 1500, 0.5, EASE_OUT)
Print shown

=== UserFolder
; Keep saves in the player's own application folder, which is always writable.
; The result ends with a slash, so add the file name straight on.
folder$ = UserFolder("My Space Game")
SetSetting "best.score", "1200"
SaveSettings folder$ + "settings.txt"

=== SetSetting
SetSetting "audio.music", 0.8
SetSetting "player.name", "Ann"
SetSetting "best.score", 12500

=== GetSetting
name$ = GetSetting("player.name", "Player")
Print "Hello " + name

=== GetSettingInt
; Show the best score, or 0 when no score has been saved yet
SetSetting "best.score", "1200"   ; normally this comes from LoadSettings
Print GetSettingInt("best.score", 0)    ; 1200
Print GetSettingInt("best.time", -1)    ; -1 (no such setting, so the default)

=== GetSettingFloat
; Music volume from the options, falling back to full volume the first time
SetSetting "audio.music", "0.75"
Print GetSettingFloat("audio.music", 1.0)    ; 0.75
Print GetSettingFloat("audio.effects", 1.0)  ; 1.0

=== SettingExists
; Offer "Continue" only when a game has been saved
SetSetting "slot1.level", "4"
If SettingExists("slot1.level") Then Print "Continue?"   ; Continue?
If Not SettingExists("slot2.level") Then Print "Slot 2 is empty"   ; Slot 2 is empty

=== DeleteSetting
; Empty a save slot
SetSetting "slot1.level", "4"
DeleteSetting "slot1.level"
Print SettingExists("slot1.level")   ; 0

=== ClearSettings
; Reset to the defaults (the saved file is untouched until SaveSettings is called)
SetSetting "audio.music", "0.2"
ClearSettings
Print GetSettingFloat("audio.music", 1.0)   ; 1.0

=== SaveSettings
; Save the options to a text file the player can read and edit
SetSetting "fullscreen", "1"
SaveSettings "settings.txt"   ; in a real game: UserFolder("My Game") + "settings.txt"
f = ReadFile("settings.txt")
Print ReadLine(f)   ; fullscreen = 1
CloseFile f
DeleteFile "settings.txt"

=== LoadSettings
; Read the options at start-up. The first run has no file, so keep the defaults
SetSetting "fullscreen", "0"
SaveSettings "settings.txt"
ClearSettings
If LoadSettings("settings.txt") Then Print "loaded"   ; loaded
DeleteFile "settings.txt"
If LoadSettings("settings.txt") = 0 Then Print "first run"   ; first run

=== JoyRumble
; a heavy thump when the player lands; JoyRumble returns 1 if a gamepad shook
landed = True
If landed Then
    If JoyRumble(0.8, 0.3, 200) = 0 Then Print "No gamepad to rumble"   ; printed when none is connected
EndIf

=== BindKey
BindKey "jump", KEY_SPACE
BindKey "jump", KEY_W              ; an action can have several keys

=== BindMouse
; Fire with the left mouse button or the space bar: one action, two inputs
BindMouse "fire", MOUSE_LEFT
BindKey "fire", KEY_SPACE
Print ActionInputs("fire")   ; mouse:1,key:57

=== BindJoy
; Jump and pause on any connected gamepad
BindJoy "jump", JOY_A
BindJoy "pause", JOY_START
Print ActionInputs("jump")   ; joy:1

=== BindJoyAxis
BindJoyAxis "left", "x", -1        ; left stick pushed left
BindJoyAxis "right", "x", 1

=== BindAction
; Let the player pick the key for "jump": LastInput gives what was pressed, in BindAction's form
Graphics 640, 480
While Not KeyHit(KEY_ESCAPE)
    pressed$ = LastInput()
    If pressed <> "" Then BindAction "jump", pressed
    Cls
    Text 10, 10, "Press anything to bind jump. Now: " + ActionInputs("jump")
    Flip
Wend

=== ClearAction
ClearAction "jump"
BindKey "jump", KEY_UP             ; the player chose a new key

=== ActionDown
BindKey "right", KEY_RIGHT
BindJoyAxis "right", "x", 1
If ActionDown("right") Then x = x + 3

=== ActionHit
BindKey "jump", KEY_SPACE
BindJoy "jump", JOY_A
If ActionHit("jump") Then Print "jump!"

=== ActionValue
; Walk slowly when the stick is pushed a little, quickly when pushed far; keys are 0 or 1
Graphics 640, 480
BindJoyAxis "right", "x", 1
BindJoyAxis "left", "x", -1
BindKey "right", KEY_RIGHT
BindKey "left", KEY_LEFT
x# = 320
While Not KeyHit(KEY_ESCAPE)
    x = x + 4 * (ActionValue("right") - ActionValue("left"))
    Cls
    Rect x, 220, 20, 20
    Flip
Wend

=== ActionInputs
BindKey "jump", KEY_SPACE
BindJoy "jump", JOY_A
Print ActionInputs("jump")         ; key:57,joy:1

=== LastInput
; "press a key for Jump": wait for something, then bind it
Graphics 640, 480
Repeat
    Flip
    pressed$ = LastInput()
Until pressed <> ""
ClearAction "jump"
BindAction "jump", pressed

=== InputName
Print InputName("key:57")          ; Space
Print InputName("joy:1")           ; Gamepad A

=== SaveActions
; Save the player's controls so they can be loaded next time with LoadActions
BindKey "jump", KEY_SPACE
SaveActions "controls.txt"   ; in a real game: UserFolder("My Game") + "controls.txt"
f = ReadFile("controls.txt")
Print ReadLine(f)   ; jump=key:57
CloseFile f
DeleteFile "controls.txt"

=== LoadActions
If LoadActions(UserFolder("My Space Game") + "controls.txt") = 0
    BindKey "jump", KEY_SPACE      ; the default controls
EndIf

=== TweenPosition
; a door slides 2 units to the right in one second, and the loop waits until it is open
Graphics3D 800, 600
cam = CreateCamera()
door = CreateCube()
PositionEntity door, 0, 0, 6
TweenPosition door, 2, 0, 6, 1        ; the target is a position, like PositionEntity
While Tweening(door)
    RenderWorld
    Flip
Wend

=== TweenRotation
; a turret swings round to face right over one and a half seconds, starting slowly and ending slowly
Graphics3D 800, 600
cam = CreateCamera()
turret = CreateCone()
PositionEntity turret, 0, 0, 6
TweenRotation turret, 0, 90, 0, 1.5, EASE_IN_OUT
While Tweening(turret)
    RenderWorld
    Flip
Wend

=== TweenScale
; a pickup pops into view
gem = CreateSphere()
ScaleEntity gem, 0, 0, 0
TweenScale gem, 1, 1, 1, 0.4, EASE_OUT_BACK

=== TweenAlpha
; a ghost fades away over two seconds, then is removed
Graphics3D 800, 600
cam = CreateCamera()
ghost = CreateSphere()
PositionEntity ghost, 0, 0, 5
TweenAlpha ghost, 0, 2                ; 0 = fully see-through
While Tweening(ghost)
    RenderWorld
    Flip
Wend
FreeEntity ghost

=== TweenColor
; a warning lamp turns from white to red in half a second
Graphics3D 800, 600
cam = CreateCamera()
lamp = CreateSphere()
PositionEntity lamp, 0, 0, 5
TweenColor lamp, 255, 40, 40, 0.5
While Not KeyHit(KEY_ESCAPE)
    RenderWorld
    Flip
Wend

=== Tweening
door = CreateCube()
TweenPosition door, 2, 0, 0, 1
While Tweening(door)
    RenderWorld
    Flip
Wend

=== StopTweens
lift = CreateCube()
TweenPosition lift, 0, 10, 0, 5
StopTweens lift                    ; the lift stops where it is

=== CameraShake
; an explosion nearby shakes the picture for 0.6 seconds
Graphics3D 800, 600
cam = CreateCamera()
CameraShake cam, 1.5, 0.6

=== ScreenFade
; fade to black, load the next level, fade back in
ScreenFade 0, 0, 0, 1, 0.5
While ScreenFading()
    RenderWorld
    Flip
Wend
ScreenFade 0, 0, 0, 0, 0.5

=== ScreenFading
ScreenFade 255, 255, 255, 0, 0.3   ; a white flash fading out
If ScreenFading() Then Print "fading"

=== CreateParticles2D
; make a spark system, give it a position and shoot 30 particles when something is hit
Graphics3D 800, 600
sparks = CreateParticles2D()
PositionParticles2D sparks, 400, 300
EmitParticles2D sparks, 30
While Not KeyHit(KEY_ESCAPE)
    Cls
    DrawParticles2D sparks
    Flip
Wend

=== PositionParticles2D
; a campfire at the bottom middle of the screen, burning continuously
Graphics 640, 480
fire = CreateParticles2D()
PositionParticles2D fire, 320, 400
Particles2DRate fire, 40
While Not KeyHit(KEY_ESCAPE)
    Cls
    DrawParticles2D fire
    Flip
Wend

=== Particles2DRate
smoke = CreateParticles2D()
Particles2DRate smoke, 30          ; 30 a second, all the time

=== EmitParticles2D
; an explosion: 200 sparks at once where the ship was
boom = CreateParticles2D()
PositionParticles2D boom, 200, 150
EmitParticles2D boom, 200

=== Particles2DLife
; short-lived dust puffs, lasting between 0.3 and 0.8 seconds
dust = CreateParticles2D()
Particles2DLife dust, 0.3, 0.8
EmitParticles2D dust, 20
Print Particles2DCount(dust)        ; 20

=== Particles2DSpeed
; a fountain: fast, upwards, in a narrow spray (-90 is up, spread of 20 degrees)
fountain = CreateParticles2D()
Particles2DSpeed fountain, 200, 300, -90, 20
Particles2DGravity fountain, 0, 400
EmitParticles2D fountain, 100
Print Particles2DCount(fountain)        ; 100

=== Particles2DGravity
fountain = CreateParticles2D()
Particles2DGravity fountain, 0, 400        ; the water falls back down

=== Particles2DSize
smoke = CreateParticles2D()
Particles2DSize smoke, 0.5, 3              ; puffs grow as they rise

=== Particles2DSpin
; falling leaves that twirl either way, up to 90 degrees a second
leaves = CreateParticles2D()
Particles2DSpin leaves, -90, 90
Particles2DGravity leaves, 0, 30
EmitParticles2D leaves, 10
Print Particles2DCount(leaves)        ; 10

=== Particles2DColor
fire = CreateParticles2D()
Particles2DColor fire, 255, 220, 80, 1     ; bright yellow ...

=== Particles2DFadeColor
fire = CreateParticles2D()
Particles2DFadeColor fire, 200, 30, 0, 0   ; ... to a fading red

=== DrawParticles2D
Graphics 640, 480
rain = CreateParticles2D()
PositionParticles2D rain, 320, -10
Particles2DRate rain, 200
Particles2DSpeed rain, 300, 400, 90, 10
While Not KeyHit(KEY_ESCAPE)
    Cls
    DrawParticles2D rain
    Flip
Wend

=== Particles2DCount
sparks = CreateParticles2D()
EmitParticles2D sparks, 50
Print Particles2DCount(sparks)

=== FreeParticles2D
; free an explosion once it has finished, so it does not pile up
boom = CreateParticles2D()
EmitParticles2D boom, 80
FreeParticles2D boom
boom = 0
