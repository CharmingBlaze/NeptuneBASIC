; Examples for the input, sound and scene commands. Format: see tools/check_doc_examples.py.

=== WaitKey
Print "Press any key to start..."
code = WaitKey()          ; pauses here until a key is pressed
Print "You pressed key code " + code

=== KeyDown
Graphics3D 640, 480
x = 0
While Not KeyDown(KEY_ESCAPE)     ; keep going until Escape is held
    If KeyDown(KEY_RIGHT) Then x = x + 1   ; true every frame the key is held
    If KeyDown(KEY_LEFT) Then x = x - 1
    Flip
Wend

=== KeyHit
Graphics3D 640, 480
jumps = 0
While Not KeyDown(KEY_ESCAPE)
    ; KeyHit counts presses since the last call, so holding the key counts once
    jumps = jumps + KeyHit(KEY_SPACE)
    Flip
Wend
Print "You jumped " + jumps + " times"

=== GetKey
Graphics3D 640, 480
typed = ""
While Not KeyDown(KEY_ENTER)
    letter = GetKey()                       ; 0 when nothing was typed
    If letter >= 32 Then typed = typed + Chr(letter)   ; keep printable characters only
    Flip
Wend
Print "You typed: " + typed

=== FlushKeys
Graphics3D 640, 480
Print "Press a key to continue"
FlushKeys                  ; ignore keys pressed before this point
code = WaitKey()

=== LoadSound
blip = LoadSound("examples/media/blip.wav")
If blip = 0 Then RuntimeError "Could not load the sound"   ; 0 means it failed
PlaySound blip

=== Load3DSound
Graphics3D 640, 480
engine = Load3DSound("examples/media/hum.wav")   ; to be played with EmitSound
If engine = 0 Then RuntimeError "Could not load the sound"

=== FreeSound
blip = LoadSound("examples/media/blip.wav")
PlaySound blip
FreeSound blip             ; the handle must not be used after this

=== LoopSound
hum = LoadSound("examples/media/hum.wav")
LoopSound hum              ; PlaySound now repeats the sound
channel = PlaySound(hum)

=== SoundVolume
blip = LoadSound("examples/media/blip.wav")
SoundVolume blip, 0.5      ; 0 = silent, 1 = full volume
PlaySound blip

=== SoundPan
blip = LoadSound("examples/media/blip.wav")
SoundPan blip, -1          ; -1 = left speaker, 0 = centre, 1 = right
PlaySound blip

=== SoundPitch
blip = LoadSound("examples/media/blip.wav")
SoundPitch blip, 22050     ; playback rate in Hz: lower is deeper and slower
PlaySound blip

=== PlaySound
blip = LoadSound("examples/media/blip.wav")
channel = PlaySound(blip)  ; the channel lets you control this playing copy
If ChannelPlaying(channel) Then Print "Playing"

=== PlayCDTrack
channel = PlayCDTrack(2)   ; CD audio is not supported: does nothing, returns 0
Print channel

=== PlayMusic
music = PlayMusic("theme.ogg", 1)   ; 1 = loop, 0 = play once
If music = 0 Then Print "Could not load the music"

=== StopChannel
hum = LoadSound("examples/media/hum.wav")
LoopSound hum
channel = PlaySound(hum)
StopChannel channel        ; silence it

=== PauseChannel
music = PlayMusic("theme.ogg", 1)
PauseChannel music         ; for example when the game is paused

=== ResumeChannel
music = PlayMusic("theme.ogg", 1)
PauseChannel music
ResumeChannel music        ; carries on from where it stopped

=== ChannelVolume
hum = LoadSound("examples/media/hum.wav")
channel = PlaySound(hum)
ChannelVolume channel, 0.25   ; 0 = silent, 1 = full; only this playing copy

=== CreateAudioBus
; one bus each for music and effects, so their volumes can be set apart
music = CreateAudioBus("music")
effects = CreateAudioBus("effects")
BusVolume music, 0.5
SoundBus LoadSound("examples/media/hum.wav"), music

=== BusVolume
; the options menu's volume sliders set the buses
effects = CreateAudioBus("effects")
BusVolume effects, 0.8     ; 0 = silent, 1 = full
BusVolume 0, 1             ; bus 0 is the master bus everything ends in

=== BusReverb
; sounds in a cave echo for three seconds
cave = CreateAudioBus("cave")
BusReverb cave, 3.0, 0.6, 6000   ; decay seconds, dry 0-1, brightness in Hz
SoundBus LoadSound("examples/media/blip.wav"), cave
BusReverb cave, 0                ; 0 removes the reverb again

=== BusFilter
; muffle everything while the player is under water
world = CreateAudioBus("world")
BusFilter world, FILTER_LOWPASS, 800     ; only sounds below about 800 Hz pass
BusFilter world, FILTER_NONE             ; back to normal

=== SetHRTF
; head-related 3D sound, if the game has the measurements
Graphics3D 800, 600
If SetHRTF("sounds/head.bin") = 0 Then DebugLog "plain panning"
SetHRTF                          ; no file: back to plain panning

=== SoundBus
; every later PlaySound of this sound goes into the voices bus
voices = CreateAudioBus("voices")
hello = LoadSound("examples/media/blip.wav")
SoundBus hello, voices
PlaySound hello

=== ChannelBus
; move a sound that is already playing into another bus
radio = CreateAudioBus("radio")
BusFilter radio, FILTER_HIGHPASS, 1500   ; thin, like a small speaker
channel = PlaySound(LoadSound("examples/media/hum.wav"))
ChannelBus channel, radio

=== ChannelPan
blip = LoadSound("examples/media/blip.wav")
channel = PlaySound(blip)
ChannelPan channel, 1      ; -1 = left, 1 = right

=== ChannelPitch
hum = LoadSound("examples/media/hum.wav")
channel = PlaySound(hum)
ChannelPitch channel, 44100   ; playback rate in Hz

=== ChannelPlaying
blip = LoadSound("examples/media/blip.wav")
channel = PlaySound(blip)
While ChannelPlaying(channel)   ; 1 while it plays, 0 once it has finished
    Delay 10
Wend
Print "Sound finished"

=== CreateListener
Graphics3D 640, 480
camera = CreateCamera()
listener = CreateListener(camera)   ; the ears follow the camera
engine = Load3DSound("examples/media/hum.wav")

=== EmitSound
Graphics3D 640, 480
camera = CreateCamera()
listener = CreateListener(camera)
engine = Load3DSound("examples/media/hum.wav")
car = CreateCube()
MoveEntity car, 5, 0, 10            ; to the right and ahead of the camera
channel = EmitSound(engine, car)    ; the sound comes from the car's position

=== LoadSoundAsync
Graphics3D 640, 480
boom = LoadSoundAsync("explosion.wav")   ; returns at once; the game keeps running
While SoundLoaded(boom) = 0
    Flip                                 ; show a loading screen meanwhile
Wend

=== SoundLoaded
Graphics3D 640, 480
boom = LoadSoundAsync("explosion.wav")
state = SoundLoaded(boom)   ; 1 = ready, 0 = still loading, -1 = failed
If state = 1 Then PlaySound boom
If state = -1 Then Print "Could not load explosion.wav"

=== SoundName
blip = LoadSound("examples/media/blip.wav")

=== SaveScene
Graphics3D 640, 480
tower = CreateCube()
ScaleEntity tower, 1, 4, 1
If SaveScene("level1.rgs") Then Print "Scene saved"   ; 1 on success

=== SaveEntity
Graphics3D 640, 480
house = CreateCube()
roof = CreateCone()
EntityParent roof, house
If SaveEntity(house, "house.rgs") Then Print "House saved"   ; includes the roof

=== ReloadAssets

=== WatchAssets
Graphics3D 640, 480
WatchAssets 1              ; edited textures and models update while the game runs

=== AssetsReloaded

changed = AssetsReloaded()   ; assets reloaded since the last call

=== EntityBatch
Graphics3D 640, 480
forest = CreateMesh()
For i = 1 To 20
    tree = CreateCube(forest)       ; children of the mesh are merged
    PositionEntity tree, i * 2, 0, 0
Next
EntityBatch forest, 1               ; 1 = merge where they are now

=== InstancePrefab
Graphics3D 640, 480
rockPrefab = CreatePrefab(CreateSphere())
For i = 1 To 5
    rock = InstancePrefab(rockPrefab)   ; a new copy each time
    PositionEntity rock, i * 3, 0, 0
Next
